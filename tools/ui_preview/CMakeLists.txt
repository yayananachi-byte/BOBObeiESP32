cmake_minimum_required(VERSION 3.16)
project(bobobei_ui_preview C CXX)
set(CMAKE_C_STANDARD 99)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
file(GLOB UI_FONTS ../../main/fonts/*.c)
add_executable(ui_preview preview.c ${UI_FONTS})
target_include_directories(ui_preview PRIVATE stubs ../../main)
target_compile_definitions(ui_preview PRIVATE _DEFAULT_SOURCE)
target_link_libraries(ui_preview PRIVATE lvgl m)
